# День 2. grep, ругулярки, find

- grep "error" file.log           # строки с error

grep -c "error" file.log        # сколько их
grep -v "INFO" file.log         # всё, КРОМЕ INFO
grep -r "password" .            # по всем файлам в папке

grep -l "error" *.log           # только имена файлов с совпадением

# День 3. Права доступка

755 = rwxr-xr-x — скрипт, папка. Стандарт для исполняемого.
644 = rw-r--r-- — документ, конфиг. Стандарт для файлов.
600 = rw------- — только владелец. Стандарт для секретов.
777 = rwxrwxrwx — ОПАСНО. Все могут всё. Никогда на продакшене.

find . -perm 777 -exec chmod 755 {} \; -> Нашли опасные права - починили

# День 4. Процессы

Процесс - это экземпляр выполняющейся программы.

ps aux - просмотреть все процессы
ps aux | grep nginx - просмотреть процессы nginx(по имени)
ps aux | sort -k3 -rn| head -10 - реальный кейс, найти топ 10 жирных процессов


top - просмотр процессов режиме реального времени

kill - мягкое завершение процесса
kill -9 жесткое завершение процесса, ядро убивает процесс мгновенно
